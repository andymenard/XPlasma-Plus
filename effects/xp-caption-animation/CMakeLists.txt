cmake_minimum_required(VERSION 3.25)
project(xp-caption-animation VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_SHARED_MODULE_PREFIX "")
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus Quick)
include(KDEInstallDirs6)
find_package(KF6 REQUIRED COMPONENTS Config CoreAddons)
find_package(KWin REQUIRED)
kcoreaddons_add_plugin(xp-caption-animation SOURCES effect.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
if(KWin_VERSION VERSION_LESS "6.6" OR NOT KWin_VERSION VERSION_LESS "6.9")
    message(FATAL_ERROR "This effect currently supports KWin 6.6 through 6.8")
endif()
if(NOT KWin_VERSION VERSION_LESS "6.8")
    target_compile_definitions(xp-caption-animation PRIVATE XP_KWIN_68)
endif()
if(KWin_VERSION VERSION_LESS "6.7")
    target_compile_definitions(xp-caption-animation PRIVATE XP_KWIN_66)
endif()
file(WRITE "${CMAKE_BINARY_DIR}/build-abi.json" "{\"kwin\":\"${KWin_VERSION}\",\"qt\":\"${Qt6Core_VERSION}\"}\n")
target_link_libraries(xp-caption-animation PRIVATE KWin::kwin Qt6::Gui KF6::ConfigCore)
add_executable(test-caption test-caption.cpp)
target_link_libraries(test-caption PRIVATE Qt6::Gui)
enable_testing()
add_test(NAME caption-motion COMMAND test-caption)
add_executable(animation-demo demo.cpp)
target_link_libraries(animation-demo PRIVATE Qt6::Widgets)
